  function makeFoundationQuestions(group){
    const rows=alphabet.filter(row=>group.letters.includes(row[0]));
    const options=(answer,values)=>shuffled([answer,...shuffled(values.filter(x=>x!==answer)).slice(0,3)]);
    const selected=shuffled(rows);
    const item=(type,uid,fields)=>({type,uid:`${group.id}-${uid}`,skill:'alphabet',...fields});
    const questions=[];
    for(let i=0;i<2;i++){
      const [letter,word]=selected[i];const lower=letter.toLowerCase();
      questions.push(item('choice',`case-${letter}`,{q:`ما شكل الحرف الصغير المقابل لـ ${letter}؟`,options:options(lower,rows.map(x=>x[0].toLowerCase())),a:lower,explain:`${letter} هو الحرف الكبير و${lower} هو الحرف الصغير.`}));
    }
    for(let i=2;i<4;i++){
      const [letter,word,ar]=selected[i];
      questions.push(item('type',`write-${letter}`,{q:`اكتب الحرف الأول من كلمة ${word} (${ar})`,answers:[letter,letter.toLowerCase()],hint:'اكتب حرفًا إنجليزيًا واحدًا',explain:`${word} تبدأ بحرف ${letter}.`}));
    }
    for(let i=4;i<6;i++){
      const [letter,word,ar]=selected[i];
      questions.push(item(('speechSynthesis' in window)?'listen':'choice',`listen-${letter}`,{q:('speechSynthesis' in window)?'استمع للكلمة، ثم اختر حرفها الأول':`ما أول حرف في كلمة ${word}؟`,say:word,options:options(letter,rows.map(x=>x[0])),a:letter,explain:`الكلمة هي ${word} (${ar})، وتبدأ بحرف ${letter}.`}));
    }
    const first=rows.slice(0,3);
    questions.push(item('match','match-1',{q:'صِل الحرف الكبير بصيغته الصغيرة',pairs:shuffled(first.map(r=>[r[0],r[0].toLowerCase()])),explain:'كل حرف إنجليزي له شكل كبير وآخر صغير.'}));
    const target=rows[rows.length-2][0],next=rows[rows.length-1][0];
    questions.push(item('choice',`order-${target}`,{q:`أي حرف يأتي بعد ${target} في الأبجدية؟`,options:options(next,rows.map(x=>x[0])),a:next,explain:`يأتي ${next} بعد ${target}.`}));
    return shuffled(questions);
  }

  function makePhonicsQuestions(g){
    const w=g.words;const questions=[];
    for(const item of w){
      const letters=[...item.word,item.extra];
      questions.push({type:'blend',q:`ركّب الكلمة التي تعني «${item.ar}» بعد الاستماع`,say:item.word,tiles:shuffled(letters),answer:item.word,skill:'alphabet',explain:`${item.word} = ${item.ar}. الأصوات: ${item.parts.join(' + ')}.`});
    }
    for(const item of w.slice(0,2)){
      questions.push({type:'listen',q:'استمع: ما الكلمة التي سمعتها؟',say:item.word,options:shuffled(w.map(z=>z.word)),a:item.word,skill:'listening',explain:`${item.word} تعني «${item.ar}».`});
    }
    const target=w[2];
    questions.push({type:'choice',q:`اختر الكلمة الإنجليزية التي تعني «${target.ar}»`,options:shuffled(w.map(z=>z.word)),a:target.word,skill:'vocab',explain:`${target.word} = ${target.ar}.`});
    const last=w[3];
    questions.push({type:'type',q:`اكتب الكلمة الإنجليزية التي تعني «${last.ar}»`,answers:[last.word],hint:`تبدأ بالحرف ${last.word[0].toUpperCase()} وتتكون من 3 أحرف`,skill:'writing',explain:`${last.word} = ${last.ar}.`});
    return prepareQuestions(questions,g.id);
  }
